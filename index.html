// Games newer than every anchor cannot prove anything, so start just above the newest
// anchor when the client's cursor is an official gameId. A miss rescans from the top.
const startCursor = (() => {
  if (bser.userGamesCursorIsGameId !== true) return undefined
  let largest = -1
  for (const gameId of anchorGameIds) {
    if (!/^[0-9]+$/.test(gameId)) return undefined
    const numeric = Number(gameId)
    if (!Number.isSafeInteger(numeric)) return undefined
    largest = Math.max(largest, numeric)
  }
  if (largest < 0) return undefined
  const cursor = largest + 1
  return Number.isSafeInteger(cursor) ? cursor : undefined
})()

LLM이 쓴 분석 문장의 숫자를 원본 데이터와 대조합니다

2026.9 · TypeScript, LLM API

LLM은 승률이나 순위가 틀려도 자연스러운 문장을 씁니다. 그래서 글을 항목별 JSON으로 쓰게 하고, 항목마다 근거로 쓴 데이터 키를 붙이게 했습니다. 검사기는 문장 속 숫자가 근거 값과 정확히 같은지 확인하고, 반올림이나 "약 몇 %"도 틀린 것으로 봅니다.

처음에는 한 곳만 틀려도 글 전체를 다시 쓰게 해서, 맞던 항목까지 바뀌었습니다. 틀린 항목만 다시 쓰게 바꿨습니다. 이때 근거 키를 바꾸면 그 결과를 버립니다. 근거까지 바꿀 수 있으면, 검사를 통과하려고 근거를 바꿀 수 있기 때문입니다. 고친 결과는 처음부터 다시 검사하고, 통과하지 못하면 내보내지 않습니다.

고친 코드 한 곳: 팀 순위를 누구 기록에서 가져올까

2026.10 · src/teams.ts · 테스트

한 경기의 참가자를 팀으로 묶는 함수입니다. 처음에는 팀 순위와 팀 킬을 첫 번째 팀원 기록에서 가져왔습니다.

실제 경기 기록을 확인하다, 게임을 포기한 플레이어 행에는 나간 시점의 값(5위, 팀 킬 14)이 남고 팀원 행에는 3위, 18이 남은 경기를 찾았습니다. 포기한 사람이 첫 번째로 오면 팀 순위가 틀리게 나옵니다. 그래서 포기하지 않은 팀원 중 가장 오래 남은 사람을 기준으로 바꿨습니다.

// A player who gave up keeps their rank and team kills from when they left,
// so team values come from someone who stayed: not given up, longest duration.
function stayScore(p: UserGame): [number, number] {
  const duration = typeof p.duration === 'number' && Number.isFinite(p.duration) ? p.duration : -1
  return [p.giveUp === 1 ? 0 : 1, duration]
}

function lastToLeave(members: UserGame[]): UserGame {
  let best = members[0]!
  let bestScore = stayScore(best)
  for (const p of members) {
    const score = stayScore(p)
    if (score[0] > bestScore[0] || (score[0] === bestScore[0] && score[1] > bestScore[1])) {
      best = p
      bestScore = score
    }
  }
  return best
}

처음 고친 버전은 남은 시간만 비교했습니다. 다른 모델에게 검토를 맡겼더니 "시간이 같거나 값이 비어 있으면 입력 순서에 따라 결과가 달라진다"는 지적이 나왔습니다. 그래서 포기 여부를 먼저 보고, 시간 값이 없으면 맨 뒤로 보내도록 고쳤습니다. 포기 여부나 남은 시간이 다른 팀원은 입력 순서가 바뀌어도 같은 사람을 고르는지 확인하는 테스트를 추가했습니다.
